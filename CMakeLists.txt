cmake_minimum_required(VERSION 3.15)
project(GrannyTacticalLab LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# ShadowHook 2.0.1: either its source tree (SHADOWHOOK_SOURCE) or the prebuilt AAR prefab
# (SHADOWHOOK_PREBUILT pointing at the unpacked AAR).
if(SHADOWHOOK_SOURCE)
  add_subdirectory(${SHADOWHOOK_SOURCE}/shadowhook/src/main/cpp shadowhook)
  target_compile_options(shadowhook PRIVATE -Wno-error -Wno-everything)
  target_link_options(shadowhook PRIVATE -Wl,--undefined-version)
  set(SHADOWHOOK_INCLUDE ${SHADOWHOOK_SOURCE}/shadowhook/src/main/cpp/include)
else()
  add_library(shadowhook SHARED IMPORTED)
  set_target_properties(shadowhook PROPERTIES IMPORTED_LOCATION ${SHADOWHOOK_PREBUILT}/prefab/modules/shadowhook/libs/android.arm64-v8a/libshadowhook.so)
  set(SHADOWHOOK_INCLUDE ${SHADOWHOOK_PREBUILT}/prefab/modules/shadowhook/include)
endif()
add_library(granny_csgo SHARED native/main.cpp)
target_include_directories(granny_csgo PRIVATE native ${GENERATED_SOURCE} ${SHADOWHOOK_INCLUDE})
target_link_libraries(granny_csgo PRIVATE shadowhook android log dl)
target_compile_options(granny_csgo PRIVATE -Wall -Wextra -Werror=return-type -fvisibility=hidden -g -fno-omit-frame-pointer)
target_link_options(granny_csgo PRIVATE -Wl,-z,max-page-size=16384)
